# M1 Data — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ingest the user's local Dukascopy EURUSD + GBPUSD 1m CSVs, build a canonical UTC Parquet store with mid + spread columns, provide multi-TF resampling (15m/1h/4h/1d), and expose a `BarProvider` + `CalendarProvider` interface the rest of the system reads from.

**Architecture:** Pure Python pipeline. CSVs → schema validation → UTC normalization → stitched monotonic 1m Parquet, partitioned by `symbol`/`year`. Resampled TFs materialized on demand via a cached helper. Calendar fetched from Forex Factory JSON feed + archived GitHub dump for history, cached per ISO-week.

**Tech Stack:** Python 3.11+, pandas 2.2+, pyarrow 16+, pydantic 2.7+, httpx, zoneinfo, typer, pytest, hypothesis, ruff, black, mypy.

**Spec:** `docs/SPEC.md` (§7 M1 Data), `docs/API.md` (§2, §5 M1), `docs/ARCHITECTURE.md` (§9).

```python
from __future__ import annotations
from pathlib import Path
from pydantic import Field
from pydantic_settings import BaseSettings, SettingsConfigDict


class Settings(BaseSettings):
    model_config = SettingsConfigDict(env_prefix="V17_", env_file=".env", extra="ignore")

    project_root: Path = Field(default_factory=lambda: Path(__file__).resolve().parents[2])
    data_root: Path | None = None
    log_level: str = "INFO"

    @property
    def data(self) -> Path:
        return self.data_root or (self.project_root / "data")

    @property
    def parquet_root(self) -> Path:
        return self.data / "parquet"

    @property
    def calendar_root(self) -> Path:
        return self.data / "calendar"

    @property
    def state_root(self) -> Path:
        return self.data / "state"

    @property
    def logs_root(self) -> Path:
        return self.project_root / "logs"

    def ensure_dirs(self) -> None:
        for p in (self.data, self.parquet_root, self.calendar_root, self.state_root, self.logs_root):
            p.mkdir(parents=True, exist_ok=True)


def get_settings() -> Settings:
    return Settings()
```

```python
from __future__ import annotations
from datetime import datetime, timezone
from decimal import Decimal
from typing import Literal
from pydantic import BaseModel, ConfigDict, field_validator, model_validator

from v17.common.types import Symbol, TF, Side


class _Frozen(BaseModel):
    model_config = ConfigDict(frozen=True, extra="forbid")


def _require_utc(ts: datetime) -> datetime:
    if ts.tzinfo is None or ts.tzinfo.utcoffset(ts) != timezone.utc.utcoffset(ts):
        raise ValueError("timestamp must be timezone-aware UTC")
    return ts


class Bar(_Frozen):
    symbol: Symbol
    tf: TF
    ts: datetime
    open: Decimal
    high: Decimal
    low: Decimal
    close: Decimal
    volume: int = 0
    spread_pips: Decimal = Decimal("0")

    @field_validator("ts")
    @classmethod
    def _ts_utc(cls, v: datetime) -> datetime:
        return _require_utc(v)

    @model_validator(mode="after")
    def _ohlc_invariants(self) -> "Bar":
        if not (self.low <= self.open <= self.high and self.low <= self.close <= self.high):
            raise ValueError("OHLC invariant violated")
        return self


class Tick(_Frozen):
    symbol: Symbol
    ts: datetime
    bid: Decimal
    ask: Decimal

    @field_validator("ts")
    @classmethod
    def _ts_utc(cls, v: datetime) -> datetime:
        return _require_utc(v)

    @model_validator(mode="after")
    def _ask_ge_bid(self) -> "Tick":
        if self.ask < self.bid:
            raise ValueError("ask must be >= bid")
        return self


class CalendarEvent(_Frozen):
    at: datetime
    currency: Literal["USD", "EUR", "GBP"]
    impact: Literal["high", "medium", "low"]
    title: str

    @field_validator("at")
    @classmethod
    def _ts_utc(cls, v: datetime) -> datetime:
        return _require_utc(v)
```

```python
from __future__ import annotations
from datetime import datetime, timezone
from typing import Protocol
from zoneinfo import ZoneInfo


class Clock(Protocol):
    def now(self) -> datetime: ...
    def now_in(self, tz: str) -> datetime: ...


class SystemClock:
    def now(self) -> datetime:
        return datetime.now(timezone.utc)

    def now_in(self, tz: str) -> datetime:
        return self.now().astimezone(ZoneInfo(tz))


class FrozenClock:
    def __init__(self, at: datetime) -> None:
        if at.tzinfo is None:
            raise ValueError("FrozenClock requires tz-aware datetime")
        self._at = at.astimezone(timezone.utc)

    def now(self) -> datetime:
        return self._at

    def now_in(self, tz: str) -> datetime:
        return self._at.astimezone(ZoneInfo(tz))
```

```python
from __future__ import annotations
from pathlib import Path
from v17.common.errors import V17DataError

DUKA_COLUMNS: tuple[str, ...] = (
    "Gmt time", "Open", "High", "Low", "Close", "Volume",
)


def parse_header(path: Path) -> list[str]:
    with path.open("r", encoding="utf-8") as fh:
        first = fh.readline().rstrip("\r\n")
    return [c.strip().strip('"') for c in first.split(",")]


def validate_schema(path: Path) -> None:
    cols = parse_header(path)
    if tuple(cols) != DUKA_COLUMNS:
        raise V17DataError("unexpected CSV header", path=str(path), got=cols, want=list(DUKA_COLUMNS))
```

```python
import pandas as pd
from typing import Literal

_TS_FMT = "%d.%m.%Y %H:%M:%S.%f"


def read_file(path: Path, side: Literal["bid", "ask"]) -> pd.DataFrame:
    validate_schema(path)
    df = pd.read_csv(path, dtype={"Open":"float64","High":"float64","Low":"float64","Close":"float64","Volume":"int64"})
    df["ts"] = pd.to_datetime(df["Gmt time"], format=_TS_FMT, utc=True)

    df = df[["ts", f"open_{side}", f"high_{side}", f"low_{side}", f"close_{side}", "Volume"]].rename(columns={"Volume":"volume"})
```

```python
# src/v17/data/parquet_store.py
from __future__ import annotations
from datetime import datetime
from pathlib import Path
import pandas as pd
import pyarrow as pa
import pyarrow.parquet as pq
from v17.common.paths import parquet_file
from v17.common.types import Symbol, TF

_SCHEMA_1M = pa.schema([
    ("ts", pa.timestamp("ns", tz="UTC")),
    ("open", pa.float64()),
    ("high", pa.float64()),
    ("low",  pa.float64()),
    ("close", pa.float64()),
    ("volume", pa.int64()),
    ("spread_pips", pa.float64()),
])


def write_1m(df: pd.DataFrame, symbol: Symbol, parquet_root: Path) -> None:
    df = df.copy()
    df["year"] = df["ts"].dt.year
    for year, chunk in df.groupby("year"):
        out = parquet_file(parquet_root, symbol, TF.M1, int(year))
        out.parent.mkdir(parents=True, exist_ok=True)
        table = pa.Table.from_pandas(chunk.drop(columns=["year"]), schema=_SCHEMA_1M, preserve_index=False)
        pq.write_table(table, out, compression="zstd")


def read_1m(symbol: Symbol, start: datetime, end: datetime, parquet_root: Path) -> pd.DataFrame:
    years = range(start.year, end.year + 1)
    frames: list[pd.DataFrame] = []
    for y in years:
        p = parquet_file(parquet_root, symbol, TF.M1, y)
        if p.exists():
            frames.append(pq.read_table(p).to_pandas())
    if not frames:
        return pd.DataFrame(columns=[f.name for f in _SCHEMA_1M])
    df = pd.concat(frames, ignore_index=True)
    mask = (df["ts"] >= pd.Timestamp(start)) & (df["ts"] <= pd.Timestamp(end))
    return df.loc[mask].sort_values("ts").reset_index(drop=True)
```

```python
# src/v17/data/resample.py
from __future__ import annotations
import pandas as pd
from v17.common.timeframes import pandas_offset
from v17.common.types import TF


def resample(df_1m: pd.DataFrame, tf: TF) -> pd.DataFrame:
    if tf == TF.M1:
        return df_1m.copy()
    offset = pandas_offset(tf)
    g = df_1m.set_index("ts").resample(offset, label="left", closed="left")
    out = pd.DataFrame({
        "open":        g["open"].first(),
        "high":        g["high"].max(),
        "low":         g["low"].min(),
        "close":       g["close"].last(),
        "volume":      g["volume"].sum(min_count=1),
        "spread_pips": g["spread_pips"].mean(),
    }).dropna(subset=["open"]).reset_index()
    out["volume"] = out["volume"].astype("int64")
    return out
```

```python
from __future__ import annotations
from datetime import datetime
from decimal import Decimal
from pathlib import Path
from typing import Iterable
import pandas as pd
from v17.common.dtos import Bar
from v17.common.types import Symbol, TF
from v17.data.parquet_store import read_1m
from v17.data.resample import resample


class ParquetBarProvider:
    def __init__(self, parquet_root: Path) -> None:
        self.parquet_root = parquet_root

    def bars(self, symbol: Symbol, tf: TF, start: datetime, end: datetime) -> Iterable[Bar]:
        df = read_1m(symbol, start, end, self.parquet_root)
        if df.empty:
            return
        if tf != TF.M1:
            df = resample(df, tf)
            df = df[(df["ts"] >= pd.Timestamp(start)) & (df["ts"] <= pd.Timestamp(end))]
        for row in df.itertuples(index=False):
            yield Bar(
                symbol=symbol, tf=tf, ts=row.ts.to_pydatetime(),
                open=Decimal(str(row.open)), high=Decimal(str(row.high)),
                low=Decimal(str(row.low)), close=Decimal(str(row.close)),
                volume=int(row.volume), spread_pips=Decimal(str(row.spread_pips)),
            )
```

```python
from __future__ import annotations
import json
from pathlib import Path
from dateutil import parser as dateparser
from v17.common.dtos import CalendarEvent

ALLOWED = {"USD", "EUR", "GBP"}
IMPACT = {"High": "high", "Medium": "medium", "Low": "low"}


def load_archive_dir(dir_: Path) -> list[CalendarEvent]:
    seen: set[tuple] = set()
    out: list[CalendarEvent] = []
    for p in sorted(dir_.glob("*.json")):
        raw = json.loads(p.read_text(encoding="utf-8"))
        for item in raw:
            if item.get("country") not in ALLOWED:
                continue
            impact = IMPACT.get(item.get("impact", ""))
            if impact is None:
                continue
            at = dateparser.isoparse(item["date"]).astimezone(__import__("datetime").timezone.utc)
            key = (at, item["country"], item.get("title",""))
            if key in seen:
                continue
            seen.add(key)
            out.append(CalendarEvent(at=at, currency=item["country"], impact=impact, title=item.get("title","")))
    out.sort(key=lambda e: e.at)
    return out
```

```python
# src/v17/data/calendar/blackout.py
from __future__ import annotations
from datetime import datetime, timedelta
from v17.common.dtos import CalendarEvent
from v17.common.types import Symbol

AFFECTED: dict[Symbol, set[str]] = {
    Symbol("EURUSD"): {"USD", "EUR"},
    Symbol("GBPUSD"): {"USD", "GBP"},
}


def is_blackout(events: list[CalendarEvent], symbol: Symbol, at: datetime, window_min: int = 5) -> bool:
    delta = timedelta(minutes=window_min)
    affected = AFFECTED.get(symbol, set())
    for e in events:
        if e.impact != "high":
            continue
        if e.currency not in affected:
            continue
        if abs((e.at - at).total_seconds()) <= delta.total_seconds():
            return True
    return False
```

```python
# src/v17/data/calendar/providers.py
from __future__ import annotations
from datetime import datetime
from pathlib import Path
import httpx
from v17.common.dtos import CalendarEvent
from v17.common.types import Symbol
from v17.data.calendar.archive import load_archive_dir
from v17.data.calendar.blackout import is_blackout as _blackout


class FFCalendarProvider:
    def __init__(self, cache_dir: Path, http: httpx.Client | None = None) -> None:
        self.cache_dir = cache_dir
        self.http = http

    def _all(self) -> list[CalendarEvent]:
        return load_archive_dir(self.cache_dir)

    def events(self, start: datetime, end: datetime) -> list[CalendarEvent]:
        return [e for e in self._all() if start <= e.at <= end]

    def is_blackout(self, symbol: Symbol, at: datetime, window_min: int = 5) -> bool:
        return _blackout(self._all(), symbol, at, window_min)
```

```python
# src/v17/data/cli.py
from __future__ import annotations
from pathlib import Path
import httpx
import typer
from v17.common.types import Symbol
from v17.common.clock import SystemClock
from v17.config import get_settings
from v17.data.dukascopy_csv import stitch_pair
from v17.data.parquet_store import write_1m
from v17.data.calendar.ff_json import fetch_current_week


def _bid_ask_paths(csv_dir: Path, symbol: str) -> tuple[list[Path], list[Path]]:
    bids = sorted(csv_dir.glob(f"{symbol}_1 Min_Bid_*.csv"))
    asks = sorted(csv_dir.glob(f"{symbol}_1 Min_Ask_*.csv"))
    if not bids or not asks:
        raise typer.BadParameter(f"no bid/ask CSVs for {symbol} under {csv_dir}")
    return bids, asks


def ingest(symbol: str = typer.Option(..., "--symbol")) -> None:
    s = get_settings(); s.ensure_dirs()
    csv_dir = s.dukascopy_csv_dir or (s.data / "raw")
    bids, asks = _bid_ask_paths(csv_dir, symbol)
    df = stitch_pair(bids, asks)
    write_1m(df, Symbol(symbol), s.parquet_root)
    typer.echo(f"wrote {len(df)} rows for {symbol} to {s.parquet_root}")


def ingest_calendar() -> None:
    s = get_settings(); s.ensure_dirs()
    with httpx.Client() as http:
        events = fetch_current_week(http, s.calendar_root, SystemClock().now())
    typer.echo(f"cached {len(events)} events to {s.calendar_root}")
```
