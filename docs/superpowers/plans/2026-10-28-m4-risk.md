# M4 Risk Engine — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:subagent-driven-development or superpowers:executing-plans.

**Goal:** Produce a `DefaultRiskEngine` that accepts a `Signal` + current `AccountState` and returns a `Decision{allow, reason, code, size_lots, sl, tp}`. The engine enforces every FTMO Challenge $100k rule in code: per-trade risk, min RR, min stop distance, max concurrent positions, max trades/day, soft daily stop + lockout, static max DD kill switch, FTMO 2026 consistency rule (best day ≤ 30% of total profit), and min trading days counter.

**Architecture:** Pure functional guards composed by `DefaultRiskEngine.check`. Each guard is a `_check_x(signal, account) -> Decision | None` that returns `None` on pass and a terminal `Decision` on reject. Order of evaluation is explicit so first failure wins. `on_fill` / `on_close` update account state immutably.

**Tech Stack:** Python 3.11+, pydantic, Decimal.

**Spec:** `docs/SPEC.md` §5 (RR), §6.3 FTMO constants, §6.4 Risk constants, §7 M4; `docs/API.md` §3 AccountState/Decision, §5 M4; `docs/decisions/0007-utc-canonical-time.md`.

```python
# src/v17/risk/sizing.py
from __future__ import annotations
from decimal import Decimal, ROUND_DOWN


def compute_lots(equity: Decimal, per_trade_pct: Decimal, stop_distance_pips: Decimal,
                 pip_value_per_lot: Decimal, lot_step: Decimal, min_lot: Decimal) -> Decimal:
    if equity <= 0 or stop_distance_pips <= 0:
        return Decimal("0")
    risk_amount = equity * per_trade_pct / Decimal("100")
    risk_per_lot = stop_distance_pips * pip_value_per_lot
    if risk_per_lot <= 0:
        return Decimal("0")
    raw = risk_amount / risk_per_lot
    floored = (raw / lot_step).to_integral_value(rounding=ROUND_DOWN) * lot_step
    if floored < min_lot:
        return Decimal("0")
    return floored.quantize(lot_step)
```

```python
from __future__ import annotations
from decimal import Decimal
from v17.common.dtos import AccountState


def would_breach_consistency(account: AccountState, est_win: Decimal,
                             best_day_pct_cap: Decimal) -> bool:
    new_today = account.daily_realized_pnl + est_win
    new_total = account.total_realized_pnl + est_win
    if new_total <= 0:
        return False
    new_best = max(account.best_day_pnl, new_today)
    return (new_best / new_total * Decimal("100")) > best_day_pct_cap
```

## Task 6: DefaultRiskEngine orchestrator

**Files:**
- Create: `src/v17/risk/engine.py`
- Test: `tests/unit/risk/test_engine.py`

**Interfaces:**
- `DefaultRiskEngine()` with `check` + `on_fill` + `on_close`.
  Check order:
  1. kill_active → reject code `kill_active`
  2. dd_breached → reject `dd_guard`
  3. daily_stop_breached → reject `daily_stop_locked`
  4. open_positions >= max_concurrent → reject `max_positions`
  5. trades_today >= max_trades_per_day → reject `max_trades_today` (needs count tracked externally; here read from account via `open_positions` + `trading_days_count` — for MVP, defer to engine param)
  6. RR < min_rr → reject `rr_too_low`
  7. stop pip distance < min_pip_sl → reject `sl_too_tight`
  8. consistency breach → reject `consistency_rule`
  9. compute size via `compute_lots`; if 0 → reject `sl_too_tight`
  10. allow with sized Decision

- `on_fill(trade, account)`: increment `open_positions`, do not change realized P&L.
- `on_close(trade, account)`: subtract from `open_positions`, add trade.pnl to daily + total; update peak_equity if equity > peak; update best_day_pnl if daily > best; if soft daily stop crossed, flip `is_locked_today = True`; if DD breached, flip `kill_active = True`.

```python
from __future__ import annotations
from decimal import Decimal
from typing import Final
from v17.common.dtos import AccountState, Decision, Signal, Trade
from v17.risk.consistency import would_breach_consistency
from v17.risk.daily_guard import daily_stop_breached
from v17.risk.dd_guard import dd_breached
from v17.risk.ftmo_rules import FTMO_CHALLENGE_100K
from v17.risk.risk_rules import RISK
from v17.risk.sizing import compute_lots


class DefaultRiskEngine:
    def __init__(
        self,
        ftmo: dict[str, object] = FTMO_CHALLENGE_100K,
        risk: dict[str, object] = RISK,
    ) -> None:
        self.ftmo = ftmo
        self.risk = risk

    def check(self, signal: Signal, account: AccountState) -> Decision:
        if account.kill_active:
            return Decision(allow=False, reason="kill switch active", code="kill_active")
        if dd_breached(account, Decimal(str(self.ftmo["max_overall_dd_pct"]))):
            return Decision(allow=False, reason="static DD breached", code="dd_guard")
        if daily_stop_breached(account, Decimal(str(self.ftmo["soft_daily_stop_pct"]))):
            return Decision(allow=False, reason="daily soft stop", code="daily_stop_locked")
        if account.open_positions >= int(self.risk["max_concurrent_positions"]):
            return Decision(allow=False, reason="max positions", code="max_positions")
        min_rr = Decimal(str(self.risk["min_rr"]))
        if signal.rr < min_rr:
            return Decision(allow=False, reason=f"rr {signal.rr} < {min_rr}", code="rr_too_low")
        pip_scale = Decimal("10000")
        stop_pips = abs(signal.entry - signal.sl) * pip_scale
        min_pip_sl = Decimal(str(self.risk["min_pip_sl"]))
        if stop_pips < min_pip_sl:
            return Decision(allow=False, reason="stop too tight", code="sl_too_tight")
        est_win = account.equity * Decimal(str(self.risk["per_trade_pct"])) / Decimal("100") * signal.rr
        if would_breach_consistency(
            account, est_win, Decimal(str(self.ftmo["consistency_best_day_pct_of_profit"]))
        ):
            return Decision(allow=False, reason="consistency rule", code="consistency_rule")
        lots = compute_lots(
            equity=account.equity,
            per_trade_pct=Decimal(str(self.risk["per_trade_pct"])),
            stop_distance_pips=stop_pips,
            pip_value_per_lot=Decimal(str(self.risk["pip_value_per_lot_usd"])),
            lot_step=Decimal(str(self.risk["lot_step"])),
            min_lot=Decimal(str(self.risk["min_lot"])),
        )
        if lots <= 0:
            return Decision(allow=False, reason="size below min lot", code="sl_too_tight")
        return Decision(
            allow=True, reason="ok", code="ok",
            size_lots=lots, sl=signal.sl, tp=signal.tp,
        )

    def on_fill(self, trade: Trade, account: AccountState) -> AccountState:
        return account.model_copy(update={"open_positions": account.open_positions + 1})

    def on_close(self, trade: Trade, account: AccountState) -> AccountState:
        pnl = trade.pnl or Decimal("0")
        new_equity = account.equity + pnl
        new_daily = account.daily_realized_pnl + pnl
        new_total = account.total_realized_pnl + pnl
        new_peak = max(account.peak_equity, new_equity)
        new_best_day = max(account.best_day_pnl, new_daily)
        soft = Decimal(str(self.ftmo["soft_daily_stop_pct"]))
        max_dd = Decimal(str(self.ftmo["max_overall_dd_pct"]))
        locked = account.is_locked_today or (new_daily / account.starting_equity * Decimal("100") <= -soft)
        kill = account.kill_active or (
            (account.starting_equity - new_equity) / account.starting_equity * Decimal("100") >= max_dd
        )
        return account.model_copy(update={
            "open_positions": max(0, account.open_positions - 1),
            "equity": new_equity,
            "balance": new_equity,
            "peak_equity": new_peak,
            "daily_realized_pnl": new_daily,
            "total_realized_pnl": new_total,
            "best_day_pnl": new_best_day,
            "is_locked_today": locked,
            "kill_active": kill,
        })
```
