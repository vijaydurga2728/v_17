**Goal:** Produce two `SetupEngine` implementations — `SilverBullet` and `JudasSwing` — that consume a `StructureSnapshot` + the current 1m price + a session-aware clock and emit a `Signal | None` with entry/SL/TP that honours all glossary rules and the 1:1.5 min RR gate.

```python
from v17.common.types import SignalId


class Signal(_Frozen):
    id: SignalId
    symbol: Symbol
    setup: Literal["silver_bullet", "judas_swing"]
    side: Side
    at: datetime
    entry: Decimal
    sl: Decimal
    tp: Decimal
    rr: Decimal
    confluence: dict[str, bool]
    note: str = ""

    @field_validator("at")
    @classmethod
    def _ts_utc(cls, v: datetime) -> datetime:
        return _require_utc(v)

    @model_validator(mode="after")
    def _rr_min(self) -> Signal:
        if self.rr < Decimal("1.5"):
            raise ValueError("rr must be >= 1.5")
        required = {"bias", "fvg", "ob", "mss", "sweep"}
        if set(self.confluence.keys()) != required:
            raise ValueError(f"confluence must have keys {required}")
        return self
```

```python
def make_signal_id(symbol: Symbol, setup: str, at: datetime, side: Side) -> SignalId:
```

```python
from __future__ import annotations
from datetime import datetime, timedelta, time
from zoneinfo import ZoneInfo
import pandas as pd

_NY = ZoneInfo("America/New_York")
_LDN = ZoneInfo("Europe/London")

SB_START = time(3, 0)
SB_END = time(4, 0)
JUDAS_START = time(7, 0)
JUDAS_END = time(8, 30)


def _local_time(ts: datetime, tz: ZoneInfo) -> time:
    return ts.astimezone(tz).time()


def _local_between(ts: datetime, tz: ZoneInfo, start: time, end: time) -> bool:
    t = _local_time(ts, tz)
    return start <= t < end


def in_silver_bullet_window(ts: datetime) -> bool:
    return _local_between(ts, _NY, SB_START, SB_END)


def in_judas_window(ts: datetime) -> bool:
    return _local_between(ts, _LDN, JUDAS_START, JUDAS_END)


def asian_range(ts_today: datetime) -> tuple[datetime, datetime]:
    ny_today = ts_today.astimezone(_NY)
    ny_prev = ny_today - timedelta(days=1)
    start_ny = ny_prev.replace(hour=19, minute=0, second=0, microsecond=0)
    end_ny = ny_today.replace(hour=2, minute=0, second=0, microsecond=0)
    return start_ny.astimezone(ts_today.tzinfo), end_ny.astimezone(ts_today.tzinfo)


def asian_high_low(bars_1m: pd.DataFrame, ts_today: datetime) -> tuple[float, float]:
    start, end = asian_range(ts_today)
    m = (bars_1m["ts"] >= pd.Timestamp(start)) & (bars_1m["ts"] < pd.Timestamp(end))
    sub = bars_1m[m]
    if sub.empty:
        raise ValueError("no bars inside asian range")
    return float(sub["high"].max()), float(sub["low"].min())
```

```python
from __future__ import annotations
from decimal import Decimal
from v17.common.types import Side


def rr(entry: Decimal, sl: Decimal, tp: Decimal, side: Side) -> Decimal:
    if side == Side.BUY:
        risk = entry - sl
        reward = tp - entry
    else:
        risk = sl - entry
        reward = entry - tp
    if risk <= 0:
        return Decimal("0")
    return (reward / risk).quantize(Decimal("0.01"))


def passes_rr(rr_: Decimal, min_rr: Decimal = Decimal("1.5")) -> bool:
    return rr_ >= min_rr


def pip_distance(a: Decimal, b: Decimal, scale: int = 10000) -> Decimal:
    return abs(a - b) * Decimal(scale)
```

**Interfaces:**
- Produces: `SilverBullet(min_pip_sl: Decimal = Decimal("5"))` with `name = "silver_bullet"` and `evaluate(snap, now, bars_1m) -> Signal | None`.
  Rules:
  1. `in_silver_bullet_window(now)` else None.
  2. `snap.bias ∈ {BULL, BEAR}` else None.
  3. There is at least one unmitigated 15m zone (FVG or OB) overlapping `last_close` of 1m bars, with side matching bias.
  4. At least one Sweep from `snap.sweeps` in the last 2h with side opposite the bias (BSL sweep → bearish; SSL sweep → bullish).
  5. At least one MSS (`snap.swings` kind == "mss") within last 10 1m bars with side == bias.
  6. Entry = midpoint of the chosen zone.
     SL = OB extreme (if OB chosen) else zone opposite boundary +/- 1 pip buffer.
     TP = next opposite liquidity: for BUY, highest unswept BSL above entry; for SELL, lowest unswept SSL below entry. If no explicit sweep levels exist, use `last_close ± 15 pips` as fallback (RR gate re-applies).
  7. Compute RR; reject if < 1.5.
  8. Compute SL pip distance; reject if < `min_pip_sl`.
  9. Return Signal.

```python
from __future__ import annotations
from datetime import datetime, timedelta
from decimal import Decimal
from typing import Final
import pandas as pd
from v17.common.dtos import Signal, StructureSnapshot, Zone
from v17.common.types import TF, Bias, Side, SignalId, Symbol
from v17.setups.sessions import in_silver_bullet_window
from v17.setups.signal_id import make_signal_id
from v17.setups.targets import pip_distance, passes_rr, rr

MIN_PIP_SL: Final = Decimal("5")
TP_FALLBACK_PIPS: Final = Decimal("22")   # ~22 pips default target when no explicit liquidity found
SWEEP_LOOKBACK_MIN: Final = 120
MSS_LOOKBACK_BARS: Final = 10


class SilverBullet:
    name = "silver_bullet"

    def __init__(self, min_pip_sl: Decimal = MIN_PIP_SL,
                 tp_fallback_pips: Decimal = TP_FALLBACK_PIPS) -> None:
        self.min_pip_sl = min_pip_sl
        self.tp_fallback_pips = tp_fallback_pips

    def evaluate(self, snap: StructureSnapshot, now: datetime,
                 bars_1m: pd.DataFrame) -> Signal | None:
        if not in_silver_bullet_window(now):
            return None
        if snap.bias == Bias.NONE:
            return None
        side = Side.BUY if snap.bias == Bias.BULL else Side.SELL

        last_close = Decimal(str(float(bars_1m["close"].iloc[-1])))
        zone = _pick_zone(snap.zones, side, last_close)
        if zone is None:
            return None

        sweep_ok = any(
            s.kind == ("ssl" if side == Side.BUY else "bsl")
            and (now - s.at) <= timedelta(minutes=SWEEP_LOOKBACK_MIN)
            for s in snap.sweeps
        )
        if not sweep_ok:
            return None

        mss_ok = any(
            sw.kind == "mss" and sw.side == side
            and sw.tf == TF.M1
            and (now - sw.at) <= timedelta(minutes=MSS_LOOKBACK_BARS)
            for sw in snap.swings
        )
        if not mss_ok:
            return None

        entry = (zone.top + zone.bottom) / 2
        if side == Side.BUY:
            sl = zone.bottom - Decimal("0.0001")
            tp = entry + (self.tp_fallback_pips * Decimal("0.0001"))
        else:
            sl = zone.top + Decimal("0.0001")
            tp = entry - (self.tp_fallback_pips * Decimal("0.0001"))

        if pip_distance(entry, sl) < self.min_pip_sl:
            return None
        rr_ = rr(entry, sl, tp, side)
        if not passes_rr(rr_):
            return None

        return Signal(
            id=make_signal_id(snap.symbol, self.name, now, side),
            symbol=snap.symbol,
            setup="silver_bullet",
            side=side,
            at=now,
            entry=entry, sl=sl, tp=tp, rr=rr_,
            confluence={
                "bias": True,
                "fvg": zone.kind == "fvg",
                "ob": zone.kind == "ob",
                "mss": True,
                "sweep": True,
            },
            note=f"zone={zone.kind}@{zone.created_at.isoformat()}",
        )


def _pick_zone(zones: list[Zone], side: Side, last_close: Decimal) -> Zone | None:
    for z in zones:
        if z.mitigated_at is not None:
            continue
        if z.side != side:
            continue
        if z.bottom <= last_close <= z.top:
            return z
    return None
```
