# M9 Observability — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:subagent-driven-development or superpowers:executing-plans.

**Goal:** Wire up the full observability stack: SQLite persistence (trades/equity/signals/errors), rotating JSONL log files, Telegram alerter (outgoing), Telegram command bot (incoming), daily markdown report, and equity curve PNG.

**Architecture:** `TradeStore` owns SQLite via SQLAlchemy ORM with Alembic migrations. `TelegramAlerter` implements the existing `Alerter` Protocol from M8 so `LiveRunner` gets real alerts with zero code changes. `TelegramCommandBot` polls for updates in a daemon thread and dispatches `/status`, `/kill`, `/report`, `/pause`, `/resume`. `ReportGenerator` queries `TradeStore` and writes markdown + matplotlib PNG. Logging file routing is added to the existing `configure()` call in `v17.common.logging`.

**Tech Stack:** Python 3.11+, `sqlalchemy>=2.0`, `alembic>=1.13`, `matplotlib>=3.9`, `apscheduler>=3.10` (already used in M8), `httpx>=0.27` (already in deps).

**Spec:** `docs/SPEC.md` §M9, §6.8 (observability), §6.9 (boundaries).

```python
from __future__ import annotations

import json
from datetime import UTC, datetime
from decimal import Decimal
from pathlib import Path

from sqlalchemy import String, create_engine as _sa_create_engine
from sqlalchemy.orm import DeclarativeBase, Mapped, Session, mapped_column

from v17.common.dtos import AccountState, Trade
from v17.common.types import Ticket


class _Base(DeclarativeBase):
    pass


class TradeRow(_Base):
    __tablename__ = "trades"
    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)
    ticket: Mapped[int] = mapped_column(unique=True)
    signal_id: Mapped[str]
    symbol: Mapped[str]
    side: Mapped[str]
    size_lots: Mapped[str]
    entry: Mapped[str]
    sl: Mapped[str]
    tp: Mapped[str]
    opened_at: Mapped[str]
    closed_at: Mapped[str | None] = mapped_column(String, nullable=True, default=None)
    close_price: Mapped[str | None] = mapped_column(String, nullable=True, default=None)
    pnl: Mapped[str | None] = mapped_column(String, nullable=True, default=None)
    reason: Mapped[str] = mapped_column(default="")


class EquityDailyRow(_Base):
    __tablename__ = "equity_daily"
    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)
    date: Mapped[str] = mapped_column(unique=True)
    equity: Mapped[str]
    balance: Mapped[str]
    daily_pnl: Mapped[str]
    total_pnl: Mapped[str]
    dd_pct: Mapped[str]
    open_positions: Mapped[int]


class SignalRow(_Base):
    __tablename__ = "signals"
    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)
    signal_id: Mapped[str] = mapped_column(unique=True)
    symbol: Mapped[str]
    setup: Mapped[str]
    side: Mapped[str]
    at: Mapped[str]
    entry: Mapped[str]
    sl: Mapped[str]
    tp: Mapped[str]
    rr: Mapped[str]
    fired: Mapped[bool]
    reject_reason: Mapped[str] = mapped_column(default="")


class ErrorRow(_Base):
    __tablename__ = "errors"
    id: Mapped[int] = mapped_column(primary_key=True, autoincrement=True)
    level: Mapped[str]
    message: Mapped[str]
    ctx: Mapped[str] = mapped_column(default="{}")
    at: Mapped[str]


def create_engine_for_path(db_path: Path) -> object:
    db_path.parent.mkdir(parents=True, exist_ok=True)
    return _sa_create_engine(f"sqlite:///{db_path}", echo=False)


class TradeStore:
    def __init__(self, engine: object) -> None:
        self._engine = engine

    def create_tables(self) -> None:
        _Base.metadata.create_all(self._engine)  # type: ignore[arg-type]

    def record_trade(self, trade: Trade) -> None:
        with Session(self._engine) as session:  # type: ignore[arg-type]
            existing = session.get(TradeRow, None)
            # check by ticket
            from sqlalchemy import select
            row = session.scalar(select(TradeRow).where(TradeRow.ticket == int(trade.ticket)))
            if row is not None:
                return
            session.add(TradeRow(
                ticket=int(trade.ticket),
                signal_id=str(trade.signal_id),
                symbol=str(trade.symbol),
                side=trade.side.value,
                size_lots=str(trade.size_lots),
                entry=str(trade.entry),
                sl=str(trade.sl),
                tp=str(trade.tp),
                opened_at=trade.opened_at.isoformat(),
            ))
            session.commit()

    def update_close(
        self,
        ticket: Ticket,
        closed_at: datetime,
        close_price: Decimal | None,
        pnl: Decimal | None,
        reason: str,
    ) -> None:
        with Session(self._engine) as session:  # type: ignore[arg-type]
            from sqlalchemy import select
            row = session.scalar(select(TradeRow).where(TradeRow.ticket == int(ticket)))
            if row is None:
                return
            row.closed_at = closed_at.isoformat()
            row.close_price = str(close_price) if close_price is not None else None
            row.pnl = str(pnl) if pnl is not None else None
            row.reason = reason
            session.commit()

    def record_equity(self, date: str, account: AccountState) -> None:
        starting = account.starting_equity
        dd_pct = (
            (starting - account.equity) / starting * Decimal("100")
            if starting > 0
            else Decimal("0")
        )
        with Session(self._engine) as session:  # type: ignore[arg-type]
            from sqlalchemy import select
            row = session.scalar(select(EquityDailyRow).where(EquityDailyRow.date == date))
            if row is None:
                row = EquityDailyRow(date=date)
                session.add(row)
            row.equity = str(account.equity)
            row.balance = str(account.balance)
            row.daily_pnl = str(account.daily_realized_pnl)
            row.total_pnl = str(account.total_realized_pnl)
            row.dd_pct = str(dd_pct)
            row.open_positions = account.open_positions
            session.commit()

    def record_error(self, level: str, message: str, ctx: dict[str, object]) -> None:
        with Session(self._engine) as session:  # type: ignore[arg-type]
            session.add(ErrorRow(
                level=level,
                message=message,
                ctx=json.dumps(ctx),
                at=datetime.now(UTC).isoformat(),
            ))
            session.commit()

    def trades_for_date(self, date: str) -> list[TradeRow]:
        with Session(self._engine) as session:  # type: ignore[arg-type]
            from sqlalchemy import select
            rows = session.scalars(
                select(TradeRow).where(TradeRow.opened_at.like(f"{date}%"))
            ).all()
            return list(rows)

    def equity_history(self) -> list[EquityDailyRow]:
        with Session(self._engine) as session:  # type: ignore[arg-type]
            from sqlalchemy import select
            return list(session.scalars(select(EquityDailyRow).order_by(EquityDailyRow.date)).all())

    def open_trades(self) -> list[TradeRow]:
        with Session(self._engine) as session:  # type: ignore[arg-type]
            from sqlalchemy import select
            return list(session.scalars(
                select(TradeRow).where(TradeRow.closed_at.is_(None))
            ).all())

    def load_account(self, starting_equity: Decimal) -> AccountState:
        open_rows = self.open_trades()
        all_history = self.equity_history()
        if all_history:
            last = all_history[-1]
            equity = Decimal(last.equity)
            balance = Decimal(last.balance)
            daily_pnl = Decimal(last.daily_pnl)
            total_pnl = Decimal(last.total_pnl)
        else:
            equity = starting_equity
            balance = starting_equity
            daily_pnl = Decimal("0")
            total_pnl = Decimal("0")
        return AccountState(
            equity=equity,
            balance=balance,
            starting_equity=starting_equity,
            peak_equity=equity,
            daily_realized_pnl=daily_pnl,
            total_realized_pnl=total_pnl,
            best_day_pnl=Decimal("0"),
            open_positions=len(open_rows),
            trading_days_count=len(all_history),
            is_locked_today=False,
            kill_active=False,
        )
```

from __future__ import annotations

import logging
import logging.handlers
from pathlib import Path

import structlog


def configure(level: str = "INFO", log_dir: Path | None = None) -> None:
    handlers: list[logging.Handler] = [logging.StreamHandler()]
    if log_dir is not None:
        log_dir.mkdir(parents=True, exist_ok=True)
        fh = logging.handlers.TimedRotatingFileHandler(
            log_dir / "v17.jsonl",
            when="midnight",
            utc=True,
            encoding="utf-8",
            backupCount=30,
        )
        handlers.append(fh)
    logging.basicConfig(level=level, format="%(message)s", handlers=handlers, force=True)
    structlog.configure(
        processors=[
            structlog.processors.add_log_level,
            structlog.processors.TimeStamper(fmt="iso", utc=True),
            structlog.processors.StackInfoRenderer(),
            structlog.processors.format_exc_info,
            structlog.processors.JSONRenderer(sort_keys=True),
        ],
        wrapper_class=structlog.make_filtering_bound_logger(logging.getLevelName(level)),
        logger_factory=structlog.PrintLoggerFactory(),
        cache_logger_on_first_use=False,
    )


def get_logger(name: str) -> structlog.stdlib.BoundLogger:
    log: structlog.stdlib.BoundLogger = structlog.get_logger(name)
    return log
```

from __future__ import annotations

import structlog

import httpx

_log = structlog.get_logger(__name__)

_BASE = "https://api.telegram.org/bot{token}/{method}"


class TelegramAlerter:
    """Implements v17.live.alerter.Alerter via Telegram Bot API."""

    def __init__(self, token: str, chat_id: str) -> None:
        self._token = token
        self._chat_id = chat_id

    def send(self, message: str, level: str = "info") -> None:
        text = f"[{level.upper()}] {message}"
        url = _BASE.format(token=self._token, method="sendMessage")
        try:
            httpx.post(url, json={"chat_id": self._chat_id, "text": text}, timeout=5.0)
        except Exception as exc:
            _log.warning("telegram_send_failed", error=str(exc))
```

from __future__ import annotations

from decimal import Decimal
from pathlib import Path
from textwrap import dedent

import matplotlib
matplotlib.use("Agg")  # non-interactive backend
import matplotlib.pyplot as plt

from v17.common.dtos import AccountState
from v17.obs.store import TradeStore


class ReportGenerator:
    def __init__(self, store: TradeStore) -> None:
        self._store = store

    def daily_markdown(self, date_str: str, account: AccountState) -> str:
        trades = self._store.trades_for_date(date_str)
        starting = account.starting_equity
        dd_pct = (
            (starting - account.equity) / starting * Decimal("100")
            if starting > 0
            else Decimal("0")
        )
        rows = "\n".join(
            f"| {t.ticket} | {t.symbol} | {t.side} | {t.entry} "
            f"| {t.close_price or '-'} | {t.pnl or '-'} | {t.reason or '-'} |"
            for t in trades
        )
        table = (
            "| Ticket | Symbol | Side | Entry | Close | PnL | Reason |\n"
            "|--------|--------|------|-------|-------|-----|--------|\n"
            + (rows if rows else "| — | — | — | — | — | — | — |")
        )
        return dedent(f"""\
            # Daily Report — {date_str}

            ## Account
            | Metric | Value |
            |--------|-------|
            | Equity | {account.equity} |
            | Balance | {account.balance} |
            | Daily P&L | {account.daily_realized_pnl} |
            | Total P&L | {account.total_realized_pnl} |
            | DD % | {dd_pct:.2f}% |

            ## Trades ({len(trades)})
            {table}
        """)

    def equity_png(self, output_path: Path) -> None:
        history = self._store.equity_history()
        dates = [r.date for r in history]
        equities = [float(r.equity) for r in history]
        fig, ax = plt.subplots(figsize=(10, 4))
        if dates:
            ax.plot(dates, equities, marker="o", linewidth=2)
            ax.set_title("Equity Curve")
            ax.set_xlabel("Date")
            ax.set_ylabel("Equity (USD)")
            plt.xticks(rotation=45)
            plt.tight_layout()
        output_path.parent.mkdir(parents=True, exist_ok=True)
        fig.savefig(output_path)
        plt.close(fig)

## Task 4: Telegram command bot

**Files:**
- Create: `src/v17/obs/bot.py`
- Create: `tests/unit/obs/test_bot.py`

**Interfaces:**
- Consumes: `read_kill`, `write_kill` (from `v17.live.kill_switch`), `ReportGenerator` (Task 3), `Clock` (common)
- Produces: `TelegramCommandBot(token, chat_id, state_root, report_generator, clock)`

**Design:** Polls `getUpdates` every 2s in a daemon thread. Only processes messages from `chat_id` (security: ignore others). Dispatches:
- `/status` → replies with `runner.state.value` if runner wired, else "No runner wired"
- `/kill` → `write_kill(state_root)`, replies "KILL sentinel written"
- `/report` → `report_generator.daily_markdown(today, account)`, sends as message
- `/pause` → writes `state_root/PAUSE` (future: runner checks this)
- `/resume` → removes `state_root/PAUSE` if present

`wire_runner(runner)` stores weak runner reference for `/status`.
`start()` launches daemon thread.
`stop()` sets `_running = False`.

```python
from __future__ import annotations

import threading
import time
from pathlib import Path
from typing import TYPE_CHECKING, Any

import httpx
import structlog

from v17.common.clock import Clock
from v17.live.kill_switch import write_kill
from v17.obs.report import ReportGenerator

if TYPE_CHECKING:
    from v17.live.runner import LiveRunner

_log = structlog.get_logger(__name__)
_BASE = "https://api.telegram.org/bot{token}/{method}"


class TelegramCommandBot:
    def __init__(
        self,
        token: str,
        chat_id: str,
        state_root: Path,
        report_generator: ReportGenerator,
        clock: Clock,
    ) -> None:
        self._token = token
        self._chat_id = chat_id
        self._state_root = state_root
        self._report_generator = report_generator
        self._clock = clock
        self._runner: LiveRunner | None = None
        self._offset = 0
        self._running = False
        self._thread: threading.Thread | None = None

    def wire_runner(self, runner: LiveRunner) -> None:
        self._runner = runner

    def start(self) -> None:
        self._running = True
        self._thread = threading.Thread(target=self._poll_loop, daemon=True, name="tg-bot")
        self._thread.start()

    def stop(self) -> None:
        self._running = False

    def _poll_loop(self) -> None:
        while self._running:
            try:
                updates = self._get_updates()
                for update in updates:
                    self._process_update(update)
                    self._offset = update.get("update_id", self._offset) + 1
            except Exception as exc:
                _log.warning("bot_poll_error", error=str(exc))
            time.sleep(2)

    def _get_updates(self) -> list[dict[str, Any]]:
        url = _BASE.format(token=self._token, method="getUpdates")
        resp = httpx.get(
            url,
            params={"offset": self._offset, "timeout": 10},
            timeout=15.0,
        )
        data = resp.json()
        return data.get("result", [])

    def _process_update(self, update: dict[str, Any]) -> None:
        msg = update.get("message", {})
        text = msg.get("text", "")
        chat_id = str(msg.get("chat", {}).get("id", ""))
        if not text.startswith("/"):
            return
        command = text.split()[0]
        self._handle_command(command, chat_id)

    def _handle_command(self, command: str, chat_id: str) -> None:
        if chat_id != self._chat_id:
            _log.warning("bot_unauthorized_chat", chat_id=chat_id)
            return
        if command == "/status":
            if self._runner is None:
                self._send_message("No runner wired")
            else:
                self._send_message(f"Runner state: {self._runner.state.value}")
        elif command == "/kill":
            write_kill(self._state_root)
            self._send_message("KILL sentinel written — runner will go IDLE on next tick")
        elif command == "/pause":
            self._state_root.mkdir(parents=True, exist_ok=True)
            (self._state_root / "PAUSE").touch()
            self._send_message("PAUSE sentinel written")
        elif command == "/resume":
            p = self._state_root / "PAUSE"
            if p.exists():
                p.unlink()
            self._send_message("PAUSE sentinel cleared")
        elif command == "/report":
            now = self._clock.now()
            date_str = now.strftime("%Y-%m-%d")
            from v17.common.dtos import AccountState
            from decimal import Decimal
            account = AccountState(
                equity=Decimal("0"), balance=Decimal("0"),
                starting_equity=Decimal("100000"), peak_equity=Decimal("0"),
                daily_realized_pnl=Decimal("0"), total_realized_pnl=Decimal("0"),
                best_day_pnl=Decimal("0"), open_positions=0,
                trading_days_count=0, is_locked_today=False, kill_active=False,
            )
            if self._runner is not None:
                pass  # M10 will wire account from runner
            md = self._report_generator.daily_markdown(date_str, account)
            self._send_message(md[:4096])  # Telegram max message length
        else:
            self._send_message(f"Unknown command: {command}. Available: /status /kill /pause /resume /report")

    def _send_message(self, text: str) -> None:
        url = _BASE.format(token=self._token, method="sendMessage")
        try:
            httpx.post(url, json={"chat_id": self._chat_id, "text": text}, timeout=5.0)
        except Exception as exc:
            _log.warning("bot_send_failed", error=str(exc))
```
