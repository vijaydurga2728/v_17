from __future__ import annotations

from decimal import Decimal
from pathlib import Path
from typing import Any, ClassVar

from pydantic import Field
from pydantic_settings import (
    BaseSettings,
    PydanticBaseSettingsSource,
    SettingsConfigDict,
    TomlConfigSettingsSource,
)


class Settings(BaseSettings):
    model_config = SettingsConfigDict(env_prefix="V17_", env_file=".env", extra="ignore")

    project_root: Path = Field(default_factory=lambda: Path(__file__).resolve().parents[2])
    data_root: Path | None = None
    dukascopy_csv_dir: Path | None = None
    log_level: str = "INFO"

    telegram_token: str = ""
    telegram_chat_id: str = ""
    starting_equity: Decimal = Decimal("100000")
    db_path: Path | None = None

    @property
    def data(self) -> Path:
        return self.data_root or (self.project_root / "data")

    @property
    def parquet_root(self) -> Path:
        return self.data / "parquet"

    @property
    def calendar_root(self) -> Path:
        return self.data / "calendar"

    @property
    def state_root(self) -> Path:
        return self.data / "state"

    @property
    def logs_root(self) -> Path:
        return self.project_root / "logs"

    @property
    def db(self) -> Path:
        return self.db_path if self.db_path is not None else self.state_root / "v17.db"

    def ensure_dirs(self) -> None:
        for p in (
            self.data,
            self.parquet_root,
            self.calendar_root,
            self.state_root,
            self.logs_root,
        ):
            p.mkdir(parents=True, exist_ok=True)

    @classmethod
    def settings_customise_sources(
        cls,
        settings_cls: type[BaseSettings],
        init_settings: PydanticBaseSettingsSource,
        env_settings: PydanticBaseSettingsSource,
        dotenv_settings: PydanticBaseSettingsSource,
        file_secret_settings: PydanticBaseSettingsSource,
    ) -> tuple[PydanticBaseSettingsSource, ...]:
        config_toml = Path(__file__).resolve().parents[2] / "config.toml"
        sources: list[PydanticBaseSettingsSource] = [
            init_settings,
            env_settings,
            dotenv_settings,
        ]
        if config_toml.exists():
            sources.append(TomlConfigSettingsSource(settings_cls, toml_file=config_toml))
        sources.append(file_secret_settings)
        return tuple(sources)


def get_settings() -> Settings:
    return Settings()

from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path

import typer

from v17.config import Settings


@dataclass
class _LiveComponents:
    runner: object  # LiveRunner — avoid circular import at module level
    store: object   # TradeStore
    bot: object | None  # TelegramCommandBot | None


def _build_components(s: Settings, dry_run: bool) -> _LiveComponents:
    from v17.common.clock import SystemClock
    from v17.common.logging import configure
    from v17.common.types import Symbol
    from v17.data.calendar.providers import FFCalendarProvider
    from v17.live.alerter import Alerter, NullAlerter
    from v17.live.runner import LiveRunner
    from v17.obs.store import TradeStore, create_engine_for_path
    from v17.risk.engine import DefaultRiskEngine
    from v17.setups.judas_swing import JudasSwing
    from v17.setups.silver_bullet import SilverBullet
    from v17.structure.engine import DefaultStructureEngine

    configure(level=s.log_level, log_dir=s.logs_root)

    engine = create_engine_for_path(s.db)
    store = TradeStore(engine)
    store.create_tables()
    account = store.load_account(starting_equity=s.starting_equity)

    alerter: Alerter
    if s.telegram_token and s.telegram_chat_id:
        from v17.obs.alerter import TelegramAlerter
        alerter = TelegramAlerter(token=s.telegram_token, chat_id=s.telegram_chat_id)
    else:
        alerter = NullAlerter()

    if dry_run:
        from v17.mt5.fake import FakeMT5
        gw = FakeMT5()
        gw.connect()
        gateway = gw
    else:
        from v17.mt5.gateway import MT5Gateway
        gw2 = MT5Gateway()
        gw2.connect()
        gateway = gw2

    clock = SystemClock()
    calendar = FFCalendarProvider(s.calendar_root)
    symbols = [Symbol("EURUSD"), Symbol("GBPUSD")]

    runner = LiveRunner(
        gateway=gateway,
        structure_engine=DefaultStructureEngine(symbols[0]),
        setup_engines=[SilverBullet(), JudasSwing()],
        risk_engine=DefaultRiskEngine(),
        calendar=calendar,
        alerter=alerter,
        state_root=s.state_root,
        symbols=symbols,
        clock=clock,
        initial_account=account,
    )

    bot = None
    if s.telegram_token and s.telegram_chat_id:
        from v17.obs.bot import TelegramCommandBot
        from v17.obs.report import ReportGenerator
        report_gen = ReportGenerator(store)
        bot = TelegramCommandBot(
            token=s.telegram_token,
            chat_id=s.telegram_chat_id,
            state_root=s.state_root,
            report_generator=report_gen,
            clock=clock,

        bot.wire_runner(runner)

    return _LiveComponents(runner=runner, store=store, bot=bot)


def paper() -> None:
    from v17.config import get_settings
    s = get_settings()
    s.ensure_dirs()
    c = _build_components(s, dry_run=True)
    if c.bot is not None:
        c.bot.start()  # type: ignore[union-attr]
    c.runner.start()  # type: ignore[union-attr]


def live(dry_run: bool = typer.Option(False, "--dry-run")) -> None:
    from v17.config import get_settings
    s = get_settings()
    s.ensure_dirs()
    if not dry_run:

            raise typer.BadParameter("GATE_PASSED sentinel missing — run 'v17 validate' first")
    c = _build_components(s, dry_run=dry_run)
    if c.bot is not None:
        c.bot.start()  # type: ignore[union-attr]
    c.runner.start()  # type: ignore[union-attr]

from __future__ import annotations

import typer


def report(day: str = typer.Option("today", "--day")) -> None:
    from datetime import date

    from v17.config import get_settings
    from v17.obs.report import ReportGenerator
    from v17.obs.store import TradeStore, create_engine_for_path

    s = get_settings()
    s.ensure_dirs()

    if day == "today":
        date_str = date.today().strftime("%Y-%m-%d")
    else:
        date_str = day

    engine = create_engine_for_path(s.db)
    store = TradeStore(engine)
    store.create_tables()
    account = store.load_account(starting_equity=s.starting_equity)
    rg = ReportGenerator(store)

    md = rg.daily_markdown(date_str, account)
    typer.echo(md)

    s.logs_root.mkdir(parents=True, exist_ok=True)
    out_md = s.logs_root / f"report-{date_str}.md"
    out_md.write_text(md)
    out_png = s.logs_root / f"equity-{date_str}.png"
    rg.equity_png(out_png)
    typer.echo(f"saved: {out_md}  equity: {out_png}")

def montecarlo_cmd(
    run: str = typer.Option(..., "--run"),
    runs: int = typer.Option(1000, "--runs"),
    seed: int = typer.Option(42, "--seed"),
) -> None:
    import json
    from decimal import Decimal

    import pandas as pd

    from v17.backtest.montecarlo import hybrid_mc

    s = get_settings()
    run_dir = s.project_root / "runs" / run
    if not run_dir.exists():
        raise typer.BadParameter(f"no run dir: {run_dir}")
    trades_path = run_dir / "trades.parquet"
    if not trades_path.exists():
        raise typer.BadParameter(f"no trades.parquet in {run_dir}")
    df = pd.read_parquet(trades_path)
    pnls = [Decimal(str(p)) for p in df["pnl"].dropna().tolist()]
    if len(pnls) < 10:
        raise typer.BadParameter(f"need ≥10 closed trades, got {len(pnls)}")
    perm = runs // 2
    boot = runs - perm
    mc = hybrid_mc(pnls, seed=seed, perm_runs=perm, boot_runs=boot, block=5)
    out = run_dir / "montecarlo.json"
    out.write_text(json.dumps(dict(mc)))
    typer.echo(
        f"p5_pf={mc['p5_pf']:.3f} p95_max_dd={mc['p95_max_dd_pct']:.2f}%  saved: {out}"
    )
