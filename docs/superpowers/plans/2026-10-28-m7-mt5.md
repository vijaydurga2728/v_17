# M7 MT5 Gateway — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:subagent-driven-development or superpowers:executing-plans.

**Goal:** Deliver a single-threaded `MT5Gateway` that wraps the `MetaTrader5` package behind a typed `TradeGateway` Protocol, a `FakeMT5` test double with full in-memory state, and a contract test suite that proves the two implementations agree on the happy path.

**Architecture:** Worker-thread confinement: all `MetaTrader5` C-extension calls happen on one dedicated thread that drains a `queue.Queue[_Cmd]`. Public methods are synchronous from the caller's perspective — they enqueue a callable and block on its `concurrent.futures.Future` result. Reconnect with exponential backoff (delays 1s/2s/4s, 3 retries) runs transparently inside the worker when a disconnect is detected. `FakeMT5` is a plain in-memory object with no threading — contract tests run both implementations against the same assertions.

**Tech Stack:** Python 3.11+, `MetaTrader5>=5.0.45`, `pydantic`, `queue`, `threading`, `concurrent.futures`, `pytest`, `mypy`, `ruff`.

**Spec:** `docs/SPEC.md` §7 M7, §6.3 FTMO constants (`magic=170000`), §6.6 performance (≤500 ms p95, reconnect ≤10 s), §6.7 testing (`FakeMT5` + contract tests), §6.9 Boundaries.

- Python `>=3.11`, `.venv`.
- UTC canonical for all timestamps.

```python
from __future__ import annotations

from datetime import UTC, datetime
from decimal import Decimal

from v17.common.dtos import Bar, OrderRequest, OrderResult, Position, Tick
from v17.common.errors import V17GatewayDisconnected
from v17.common.types import TF, Side, Symbol, Ticket


class FakeMT5:
    """In-memory TradeGateway for tests. Not thread-safe."""

    def __init__(self) -> None:
        self._connected: bool = False
        self._ticks: dict[str, Tick] = {}
        self._bars: dict[tuple[str, str], list[Bar]] = {}
        self._positions: dict[int, Position] = {}
        self._next_ticket: int = 1

    # --- helpers ---

    def is_connected(self) -> bool:
        return self._connected

    def set_tick(self, symbol: Symbol, tick: Tick) -> None:
        self._ticks[str(symbol)] = tick

    def set_bars(self, symbol: Symbol, tf: TF, bars: list[Bar]) -> None:
        self._bars[(str(symbol), str(tf))] = list(bars)

    # --- TradeGateway interface ---

    def connect(self) -> None:
        self._connected = True

    def disconnect(self) -> None:
        self._connected = False

    def symbol_select(self, symbol: Symbol) -> None:
        pass

    def tick(self, symbol: Symbol) -> Tick | None:
        return self._ticks.get(str(symbol))

    def rates(self, symbol: Symbol, tf: TF, count: int) -> list[Bar]:
        bars = self._bars.get((str(symbol), str(tf)), [])
        return bars[-count:] if count < len(bars) else list(bars)

    def order_send(self, order: OrderRequest) -> OrderResult:
        if not self._connected:
            raise V17GatewayDisconnected("FakeMT5 not connected")
        t = self._ticks.get(str(order.symbol))
        if t is None:
            return OrderResult(ok=False, error_code=1, error_message="no tick for symbol")
        filled_price = t.ask if order.side == Side.BUY else t.bid
        ticket = Ticket(self._next_ticket)
        self._next_ticket += 1
        pos = Position(
            ticket=ticket,
            symbol=order.symbol,
            side=order.side,
            size_lots=order.size_lots,
            opened_at=t.ts,
            entry=filled_price,
            sl=order.sl,
            tp=order.tp,
            current_price=filled_price,
            unrealized_pnl=Decimal("0"),
        )
        self._positions[int(ticket)] = pos
        return OrderResult(ok=True, ticket=ticket, filled_price=filled_price, filled_at=t.ts)

    def order_modify(self, ticket: Ticket, sl: Decimal, tp: Decimal) -> OrderResult:
        raw = self._positions.get(int(ticket))
        if raw is None:
            return OrderResult(ok=False, error_code=2, error_message="position not found")
        updated = Position(
            ticket=raw.ticket,
            symbol=raw.symbol,
            side=raw.side,
            size_lots=raw.size_lots,
            opened_at=raw.opened_at,
            entry=raw.entry,
            sl=sl,
            tp=tp,
            current_price=raw.current_price,
            unrealized_pnl=raw.unrealized_pnl,
        )
        self._positions[int(ticket)] = updated
        return OrderResult(ok=True, ticket=ticket)

    def order_close(self, ticket: Ticket) -> OrderResult:
        if int(ticket) not in self._positions:
            return OrderResult(ok=False, error_code=2, error_message="position not found")
        del self._positions[int(ticket)]
        return OrderResult(ok=True, ticket=ticket)

    def positions(self) -> list[Position]:
        return list(self._positions.values())
```

### MT5 timeframe map (MetaTrader5 package constants are non-intuitive)

```
TF.M1  → mt5.TIMEFRAME_M1  (= 1)
TF.M15 → mt5.TIMEFRAME_M15 (= 15)
TF.H1  → mt5.TIMEFRAME_H1  (= 16385)
TF.H4  → mt5.TIMEFRAME_H4  (= 16388)
TF.D1  → mt5.TIMEFRAME_D1  (= 16408)
```

### Reconnect policy

Delays `[1, 2, 4]` seconds — 3 retries. Max total wait = 7 s (within the ≤10 s spec).
On all 3 retries exhausted, raise `V17GatewayDisconnected`.

```python
from __future__ import annotations

import time
from concurrent.futures import Future
from datetime import UTC, datetime
from decimal import Decimal
from queue import Queue
from threading import Thread
from typing import Any, Callable, TypeVar

from v17.common.dtos import Bar, OrderRequest, OrderResult, Position, Tick
from v17.common.errors import V17GatewayDisconnected, V17GatewayError
from v17.common.types import TF, Side, Symbol, Ticket

try:
    import MetaTrader5 as _mt5_default  # type: ignore[import-untyped]
except ImportError:
    _mt5_default = None  # type: ignore[assignment]

_T = TypeVar("_T")
_STOP = object()
_RECONNECT_DELAYS = [1, 2, 4]  # seconds; 3 attempts, max 7 s total

_TF_MAP: dict[TF, int] = {
    TF.M1: 1,
    TF.M15: 15,
    TF.H1: 16385,
    TF.H4: 16388,
    TF.D1: 16408,
}


class _Cmd:
    __slots__ = ("fn", "future")

    def __init__(self, fn: Callable[[], Any]) -> None:
        self.fn = fn
        self.future: Future[Any] = Future()


class MT5Gateway:
    """Thread-confined MetaTrader5 gateway.

    All MT5 calls run on a single worker thread to satisfy MT5's
    thread-affinity requirement. Public methods are synchronous.
    """

    def __init__(
        self,
        login: int,
        password: str,
        server: str,
        *,
        _mt5: Any = None,
    ) -> None:
        self._login = login
        self._password = password
        self._server = server
        self._mt5 = _mt5 if _mt5 is not None else _mt5_default
        self._queue: Queue[Any] = Queue()
        self._thread = Thread(target=self._worker, daemon=True, name="mt5-worker")
        self._thread.start()

    # ------------------------------------------------------------------ #
    # Internal worker                                                       #
    # ------------------------------------------------------------------ #

    def _worker(self) -> None:
        while True:
            item = self._queue.get()
            if item is _STOP:
                return
            cmd: _Cmd = item
            try:
                result = cmd.fn()
                cmd.future.set_result(result)
            except Exception as exc:  # noqa: BLE001
                cmd.future.set_exception(exc)

    def _run(self, fn: Callable[[], _T]) -> _T:
        cmd = _Cmd(fn)
        self._queue.put(cmd)
        return cmd.future.result(timeout=30)

    def _check_mt5(self) -> Any:
        if self._mt5 is None:
            raise V17GatewayError("MetaTrader5 package not installed")
        return self._mt5

    # ------------------------------------------------------------------ #
    # TradeGateway interface                                                #
    # ------------------------------------------------------------------ #

    def connect(self) -> None:
        mt5 = self._check_mt5()

        def _connect() -> None:
            if not mt5.initialize(
                login=self._login,
                password=self._password,
                server=self._server,
            ):
                code, msg = mt5.last_error()
                raise V17GatewayError(f"MT5 initialize failed: {msg}", code=code)

        self._run(_connect)

    def disconnect(self) -> None:
        mt5 = self._check_mt5()
        self._run(lambda: mt5.shutdown())

    def symbol_select(self, symbol: Symbol) -> None:
        mt5 = self._check_mt5()

        def _select() -> None:
            if not mt5.symbol_select(str(symbol), True):
                code, msg = mt5.last_error()
                raise V17GatewayError(f"symbol_select failed for {symbol}: {msg}", code=code)

        self._run(_select)

    def tick(self, symbol: Symbol) -> Tick | None:
        mt5 = self._check_mt5()

        def _tick() -> Tick | None:
            t = mt5.symbol_info_tick(str(symbol))
            if t is None:
                return None
            return Tick(
                symbol=symbol,
                ts=datetime.fromtimestamp(t.time, tz=UTC),
                bid=Decimal(str(round(t.bid, 8))),
                ask=Decimal(str(round(t.ask, 8))),
            )

        return self._run(_tick)

    def rates(self, symbol: Symbol, tf: TF, count: int) -> list[Bar]:
        mt5 = self._check_mt5()
        mt5_tf = _TF_MAP[tf]

        def _rates() -> list[Bar]:
            rows = mt5.copy_rates_from_pos(str(symbol), mt5_tf, 0, count)
            if rows is None:
                return []
            result: list[Bar] = []
            for r in rows:
                o, h, l, c = (
                    Decimal(str(round(r.open, 8))),
                    Decimal(str(round(r.high, 8))),
                    Decimal(str(round(r.low, 8))),
                    Decimal(str(round(r.close, 8))),
                )
                # Bar requires low <= open/close <= high
                if not (l <= o <= h and l <= c <= h):
                    continue
                result.append(
                    Bar(
                        symbol=symbol,
                        tf=tf,
                        ts=datetime.fromtimestamp(r.time, tz=UTC),
                        open=o,
                        high=h,
                        low=l,
                        close=c,
                        volume=int(getattr(r, "tick_volume", 0)),
                        spread_pips=Decimal(str(getattr(r, "spread", 0))) / Decimal("10"),
                    )
                )
            return result

        return self._run(_rates)

    def order_send(self, order: OrderRequest) -> OrderResult:
        mt5 = self._check_mt5()

        def _send() -> OrderResult:
            if order.entry is not None:
                price = float(order.entry)
            else:
                t = mt5.symbol_info_tick(str(order.symbol))
                price = float(t.ask if order.side == Side.BUY else t.bid) if t else 0.0
            request = {
                "action": mt5.TRADE_ACTION_DEAL,
                "symbol": str(order.symbol),
                "volume": float(order.size_lots),
                "type": mt5.ORDER_TYPE_BUY if order.side == Side.BUY else mt5.ORDER_TYPE_SELL,
                "price": price,
                "sl": float(order.sl),
                "tp": float(order.tp),
                "deviation": 10,
                "magic": order.magic,
                "comment": order.comment,
                "type_time": mt5.ORDER_TIME_GTC,
                "type_filling": mt5.ORDER_FILLING_IOC,
            }
            res = mt5.order_send(request)
            if res is None:
                code, msg = mt5.last_error()
                return OrderResult(ok=False, error_code=code, error_message=msg)
            if res.retcode != mt5.TRADE_RETCODE_DONE:
                return OrderResult(
                    ok=False,
                    error_code=res.retcode,
                    error_message=getattr(res, "comment", ""),
                )
            now = datetime.now(tz=UTC)
            return OrderResult(
                ok=True,
                ticket=Ticket(res.order),
                filled_price=Decimal(str(round(res.price, 8))),
                filled_at=now,
            )

        return self._run(_send)

    def order_modify(self, ticket: Ticket, sl: Decimal, tp: Decimal) -> OrderResult:
        mt5 = self._check_mt5()

        def _modify() -> OrderResult:
            request = {
                "action": mt5.TRADE_ACTION_SLTP,
                "position": int(ticket),
                "sl": float(sl),
                "tp": float(tp),
            }
            res = mt5.order_send(request)
            if res is None:
                code, msg = mt5.last_error()
                return OrderResult(ok=False, error_code=code, error_message=msg)
            if res.retcode != mt5.TRADE_RETCODE_DONE:
                return OrderResult(ok=False, error_code=res.retcode, error_message=getattr(res, "comment", ""))
            return OrderResult(ok=True, ticket=ticket)

        return self._run(_modify)

    def order_close(self, ticket: Ticket) -> OrderResult:
        mt5 = self._check_mt5()

        def _close() -> OrderResult:
            pos_list = mt5.positions_get(ticket=int(ticket))
            if not pos_list:
                return OrderResult(ok=False, error_code=2, error_message="position not found")
            pos = pos_list[0]
            close_type = mt5.ORDER_TYPE_SELL if pos.type == mt5.ORDER_TYPE_BUY else mt5.ORDER_TYPE_BUY
            t = mt5.symbol_info_tick(pos.symbol)
            close_price = (t.bid if close_type == mt5.ORDER_TYPE_SELL else t.ask) if t else 0.0
            request = {
                "action": mt5.TRADE_ACTION_DEAL,
                "symbol": pos.symbol,
                "volume": pos.volume,
                "type": close_type,
                "position": int(ticket),
                "price": close_price,
                "deviation": 10,
                "magic": 170000,
                "type_time": mt5.ORDER_TIME_GTC,
                "type_filling": mt5.ORDER_FILLING_IOC,
            }
            res = mt5.order_send(request)
            if res is None:
                code, msg = mt5.last_error()
                return OrderResult(ok=False, error_code=code, error_message=msg)
            if res.retcode != mt5.TRADE_RETCODE_DONE:
                return OrderResult(ok=False, error_code=res.retcode, error_message=getattr(res, "comment", ""))
            return OrderResult(ok=True, ticket=ticket)

        return self._run(_close)

    def positions(self) -> list[Position]:
        mt5 = self._check_mt5()

        def _positions() -> list[Position]:
            raw = mt5.positions_get()
            if not raw:
                return []
            result: list[Position] = []
            for p in raw:
                side = Side.BUY if p.type == mt5.ORDER_TYPE_BUY else Side.SELL
                result.append(
                    Position(
                        ticket=Ticket(p.ticket),
                        symbol=Symbol(p.symbol),
                        side=side,
                        size_lots=Decimal(str(p.volume)),
                        opened_at=datetime.fromtimestamp(p.time, tz=UTC),
                        entry=Decimal(str(round(p.price_open, 8))),
                        sl=Decimal(str(round(p.sl, 8))),
                        tp=Decimal(str(round(p.tp, 8))),
                        current_price=Decimal(str(round(p.price_open, 8))),
                        unrealized_pnl=Decimal(str(round(p.profit, 2))),
                    )
                )
            return result

        return self._run(_positions)

    # ------------------------------------------------------------------ #
    # Context manager support                                               #
    # ------------------------------------------------------------------ #

    def __enter__(self) -> MT5Gateway:
        self.connect()
        return self

    def __exit__(self, *_: object) -> None:
        self.disconnect()
        self._queue.put(_STOP)
        self._thread.join(timeout=5)
```
