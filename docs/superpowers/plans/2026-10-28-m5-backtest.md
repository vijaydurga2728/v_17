# M5 Backtest Engine — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:subagent-driven-development or superpowers:executing-plans.

**Goal:** Produce a deterministic, event-driven, single-pair backtester that iterates 1m bars, invokes M2+M3+M4 at each bar, simulates realistic fills with cost model, tracks equity + trade list, and returns a `RunReport` with PF/Sharpe/winrate/max-DD/net-PnL. Add walk-forward orchestrator (12 anchored folds IS/OOS 60/40) and Monte Carlo evaluator (500 trade-order permutations + 500 block-bootstrap b=5). Write artifacts to `runs/<run_id>/`.

**Architecture:** Point-in-time snapshot built by precomputing structure once over the full slice, then filtering by `at` at each 1m bar. Fill simulator uses next-bar open ± slippage for entry, intrabar SL/TP touches respecting gap-opens. Equity tracked per close. Metrics computed from trade list and daily equity curve.

**Tech Stack:** Python 3.11+, pandas, numpy, pydantic, pyarrow (Parquet artifacts), hashlib (deterministic run_id).

**Spec:** `docs/SPEC.md` §7 M5, §6.5 cost model, §6.2 validation thresholds, §7 M6 (gate — separate module); `docs/API.md` §5 M5; `docs/decisions/0001-custom-event-driven-backtest.md`, `0003-walk-forward-anchored-12-folds.md`, `0004-monte-carlo-hybrid.md`.

```python
# src/v17/backtest/cost.py
from __future__ import annotations
from decimal import Decimal
from v17.common.types import Side, Symbol

_PIP = Decimal("0.0001")

COST_MODEL: dict[str, dict[str, Decimal]] = {
    "EURUSD": {"spread_pips": Decimal("0.8"), "slippage_pips_per_side": Decimal("0.3"),
               "commission_per_lot": Decimal("0")},
    "GBPUSD": {"spread_pips": Decimal("1.2"), "slippage_pips_per_side": Decimal("0.3"),
               "commission_per_lot": Decimal("0")},
}


def _cfg(symbol: Symbol) -> dict[str, Decimal]:
    return COST_MODEL[symbol]


def apply_entry_slippage(side: Side, price: Decimal, symbol: Symbol) -> Decimal:
    slip = _cfg(symbol)["slippage_pips_per_side"] * _PIP
    return price + slip if side == Side.BUY else price - slip


def apply_exit_slippage(side: Side, price: Decimal, symbol: Symbol) -> Decimal:
    slip = _cfg(symbol)["slippage_pips_per_side"] * _PIP
    return price - slip if side == Side.BUY else price + slip


def half_spread(symbol: Symbol) -> Decimal:
    return _cfg(symbol)["spread_pips"] * _PIP / Decimal("2")


def commission(size_lots: Decimal, symbol: Symbol) -> Decimal:
    return _cfg(symbol)["commission_per_lot"] * size_lots
```

- Create: `src/v17/backtest/fills.py`

**Interfaces:**
- `entry_fill(side, requested_entry, next_bar_open, symbol) -> Decimal` — if signal is market, fill at `next_bar_open ± slippage ± half_spread`. For simplicity we treat all entries as market-at-next-bar-open with slippage + half-spread.
- `exit_fill_on_bar(side, sl, tp, bar) -> tuple[Literal["sl","tp","none"], Decimal | None]` — intrabar hit detection. Rules:
  - BUY: if `bar.low <= sl` → SL hit at `min(bar.open, sl)` (gap honored). Elif `bar.high >= tp` → TP hit at `max(bar.open, tp)` (reverse gap).
  - SELL: mirror.
  - Both hit same bar: assume SL first (pessimistic).
- `pnl_usd(side, entry, exit_price, size_lots, pip_value_per_lot) -> Decimal` — `(exit−entry)*scale*pip_value*lots` (BUY) or inverse for SELL.

- [ ] **Step 1: Write failing tests** including the pessimistic SL-first rule for same-bar hit + the gap-honoring behavior.

```python
# src/v17/backtest/fills.py
from __future__ import annotations
from decimal import Decimal
from typing import Literal
import pandas as pd
from v17.common.types import Side, Symbol
from v17.backtest.cost import apply_entry_slippage, apply_exit_slippage, half_spread

_PIP_SCALE = Decimal("10000")


def entry_fill(side: Side, next_bar_open: Decimal, symbol: Symbol) -> Decimal:
    base = next_bar_open + half_spread(symbol) if side == Side.BUY else next_bar_open - half_spread(symbol)
    return apply_entry_slippage(side, base, symbol)


def exit_fill_on_bar(side: Side, sl: Decimal, tp: Decimal, bar: pd.Series) -> tuple[Literal["sl","tp","none"], Decimal | None]:
    bar_open = Decimal(str(float(bar["open"])))
    bar_high = Decimal(str(float(bar["high"])))
    bar_low = Decimal(str(float(bar["low"])))
    if side == Side.BUY:
        if bar_low <= sl:
            return "sl", min(bar_open, sl)
        if bar_high >= tp:
            return "tp", max(bar_open, tp)
    else:
        if bar_high >= sl:
            return "sl", max(bar_open, sl)
        if bar_low <= tp:
            return "tp", min(bar_open, tp)
    return "none", None


def pnl_usd(side: Side, entry: Decimal, exit_price: Decimal, size_lots: Decimal,
            pip_value_per_lot: Decimal = Decimal("10"), symbol: Symbol | None = None) -> Decimal:
    _ = symbol
    if side == Side.BUY:
        pips = (exit_price - entry) * _PIP_SCALE
    else:
        pips = (entry - exit_price) * _PIP_SCALE
    return pips * pip_value_per_lot * size_lots
```

```python
# src/v17/backtest/metrics.py
from __future__ import annotations
from decimal import Decimal
import math
import pandas as pd


def profit_factor(trades: list[Decimal]) -> float:
    wins = sum((t for t in trades if t > 0), start=Decimal("0"))
    losses = sum((-t for t in trades if t < 0), start=Decimal("0"))
    if losses == 0:
        return 0.0 if wins == 0 else float("inf")
    return float(wins / losses)


def winrate(trades: list[Decimal]) -> float:
    if not trades:
        return 0.0
    return sum(1 for t in trades if t > 0) / len(trades)


def max_drawdown_pct(equity: list[Decimal]) -> float:
    peak = equity[0] if equity else Decimal("0")
    worst = 0.0
    for e in equity:
        peak = max(peak, e)
        if peak > 0:
            dd = float((peak - e) / peak * 100)
            worst = max(worst, dd)
    return worst


def sharpe(daily_returns: list[float], periods_per_year: int = 252) -> float:
    if len(daily_returns) < 2:
        return 0.0
    mean = sum(daily_returns) / len(daily_returns)
    var = sum((x - mean) ** 2 for x in daily_returns) / (len(daily_returns) - 1)
    std = math.sqrt(var)
    if std == 0:
        return 0.0
    return mean / std * math.sqrt(periods_per_year)


def daily_returns_from_equity(curve: pd.DataFrame) -> list[float]:
    eq = curve["equity"].astype(float).to_numpy()
    if len(eq) < 2:
        return []
    rets = (eq[1:] - eq[:-1]) / eq[:-1]
    return [float(x) for x in rets]
```

**Interfaces:**
- `EventLoop(symbol, setups: list[SetupEngine], risk: RiskEngine, pre: PrecomputedStructure)` with `run(start, end, starting_equity, seed) -> tuple[list[Trade], pd.DataFrame]` (trades + equity curve indexed by date).

Loop pseudo:
```
account = fresh(starting_equity)
open_trade = None
for i, bar in enumerate(bars_1m within [start, end]):
    now = bar.ts
    if open_trade:
        kind, px = exit_fill_on_bar(side, sl, tp, bar)
        if kind != "none":
            close trade; account = risk.on_close(trade, account)
            record pnl, update equity for this day
            open_trade = None
        continue
    if account.is_locked_today or account.kill_active:
        continue
    snap = snapshot_at(pre, now, symbol)
    for setup in setups:
        sig = setup.evaluate(snap, now, bars_1m[:i+1])
        if sig is None: continue
        decision = risk.check(sig, account)
        if not decision.allow: continue
        # open at next-bar open ± slippage
        if i+1 >= len(bars_1m): break
        next_bar = bars_1m.iloc[i+1]
        entry = entry_fill(sig.side, Decimal(str(float(next_bar.open))), symbol)
        trade = Trade(... entry=entry, sl=sig.sl, tp=sig.tp,
                      size_lots=decision.size_lots, opened_at=next_bar.ts, ...)
        account = risk.on_fill(trade, account)
        open_trade = trade
        break  # only one at a time
# end-of-day: when date changes, reset is_locked_today, increment trading_days_count
```
